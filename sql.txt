  ### 3. SQL Server Migration Script

  Run the following SQL script on your SQL Server to add the new columns and index to your active TenantSessions
  table:

    USE [TeamsKeepAliveDb];
    GO

    -- 1. Add LocalAccountId column if it does not exist
    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID(N'[dbo].[TenantSessions]') AND name =
  'LocalAccountId')
    BEGIN
        ALTER TABLE [dbo].[TenantSessions] ADD [LocalAccountId] NVARCHAR(256) NULL;
    END
    GO

    -- 2. Add ClientId column if it does not exist
    IF NOT EXISTS (SELECT * FROM sys.columns WHERE object_id = OBJECT_ID(N'[dbo].[TenantSessions]') AND name =
  'ClientId')
    BEGIN
        ALTER TABLE [dbo].[TenantSessions] ADD [ClientId] NVARCHAR(256) NULL;
    END
    GO

    -- 3. Create index on LocalAccountId for fast query lookups
    IF NOT EXISTS (SELECT * FROM sys.indexes WHERE object_id = OBJECT_ID(N'[dbo].[TenantSessions]') AND name =
  'IX_TenantSessions_LocalAccountId')
    BEGIN
        CREATE NONCLUSTERED INDEX [IX_TenantSessions_LocalAccountId]
            ON [dbo].[TenantSessions] ([LocalAccountId] ASC);
    END
    GO
